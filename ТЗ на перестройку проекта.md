def parse_collections_file(filepath: str) -> Dict[str, dict]:
    """Парсинг файла 'Сборы - ММ 2026.xlsx'"""
    df = pd.read_excel(filepath)
    result = {}
    for _, row in df.iterrows():
        emp_name = normalize_name(row['ФИО'])
        result[emp_name] = {
            'collected_amount': float(row.get('Сумма сбора', 0)),
            'collected_count': int(row.get('Кол-во платежей', 0)),
            'plan_amount': float(row.get('План', 0)),
            'collection_rate': calculate_rate(row.get('Сумма сбора', 0), row.get('План', 1)),
            'plan_fulfillment': calculate_fulfillment(row.get('Сумма сбора', 0), row.get('План', 1))
        }
    return result

def parse_crm_actions_file(filepath: str) -> Dict[str, dict]:
    """Парсинг файла 'CRM-действия - ММ 2026.xlsx'"""
    df = pd.read_excel(filepath)
    result = {}
    for emp_name, group in df.groupby('ФИО'):
        actions = group['Тип действия'].value_counts().to_dict()
        tasks_created = len(group[group['Статус'] == 'Создана'])
        tasks_completed = len(group[group['Статус'] == 'Завершена'])
        
        # Генерация текстового вывода о поведении
        behavior_summary = generate_crm_summary(actions, tasks_completed, tasks_created)
        pattern = identify_behavior_pattern(actions)
        
        result[normalize_name(emp_name)] = {
            'crm_actions': actions,
            'crm_actions_summary': behavior_summary,
            'crm_behavior_pattern': pattern,
            'tasks_metrics': {
                'created': tasks_created,
                'completed': tasks_completed,
                'completion_rate': tasks_completed / max(tasks_created, 1)
            }
        }
    return result

def generate_crm_summary(actions: dict, completed: int, created: int) -> str:
    """Генерация человеко-читаемого вывода по CRM-логике"""
    if actions.get('refused_payment', 0) > actions.get('promised_to_pay', 0) * 2:
        return "Преобладает пассивная тактика: частые отказы без попытки переговоров"
    elif actions.get('promised_to_pay', 0) / max(completed, 1) > 0.7:
        return "Эффективная конверсия обещаний в платежи: высокая настойчивость"
    elif created > completed * 1.5:
        return "Много созданных задач с низкой завершаемостью: требуется контроль дисциплины"
    else:
        return "Сбалансированная рабочая логика"

def identify_behavior_pattern(actions: dict) -> str:
    """Определение паттерна поведения оператора"""
    total = sum(actions.values())
    if total == 0: return "unknown"
    
    if actions.get('promised_to_pay', 0) / total > 0.4:
        return "negotiator"  # Переговорщик
    elif actions.get('refused_payment', 0) / total > 0.3:
        return "passive"     # Пассивный
    elif actions.get('hung_up', 0) + actions.get('silence', 0) > total * 0.25:
        return "avoidant"    # Избегающий
    else:
        return "balanced"    # Сбалансированный

```python
# app.py - вспомогательные функции для корреляций

def calculate_efficiency_ratio(record) -> float:
    """
    Композитный показатель: как качество разговора влияет на сборы
    Формула: (TQS нормализованный) × (collection_rate) × (коэффициент активности)
    """
    tqs_norm = min(record.tqs / 100, 1.0) if record.tqs else 0
    rate_norm = min(record.collection_rate / 100, 1.0) if record.collection_rate else 0
    activity_factor = min(record.activity_count / 100, 1.5) if record.activity_count else 1
    
    return round(tqs_norm * rate_norm * activity_factor * 100, 2)

def calculate_correlation(data: list, field_x: str, field_y: str) -> float:
    """Расчет коэффициента корреляции Пирсона для вложенных полей"""
    def get_nested(obj, path: str):
        for key in path.split('.'):
            obj = obj.get(key, 0) if isinstance(obj, dict) else getattr(obj, key, 0)
        return obj
    
    x_vals = [get_nested(d, field_x) for d in data]
    y_vals = [get_nested(d, field_y) for d in data]
    
    # Удаление нулевых/пустых значений
    pairs = [(x, y) for x, y in zip(x_vals, y_vals) if x and y]
    if len(pairs) < 3:
        return 0.0
    
    from scipy import stats
    correlation, _ = stats.pearsonr([p[0] for p in pairs], [p[1] for p in pairs])
    return round(correlation, 3)
```
